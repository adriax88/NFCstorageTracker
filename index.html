function saveNewItem() { const nameInput = document.getElementById('newItemName'); const name = nameInput ? nameInput.value.trim() : ''; if (!name) { alert('Please enter an item name'); return; } const bin = bins.find(b => b.id === currentBinId); if (!bin) return; if (!bin.items) bin.items = []; const saveBtn = document.getElementById('saveItemBtn'); saveBtn.innerText = 'Compressing...'; saveBtn.disabled = true; const cameraInput = document.getElementById('newItemPhotoCamera'); const galleryInput = document.getElementById('newItemPhotoGallery'); const dateStr = new Date().toLocaleDateString(); const photoFile = (cameraInput && cameraInput.files && cameraInput.files[0]) || (galleryInput && galleryInput.files && galleryInput.files[0]); const finishSave = (photo) => { // 1. Update the data if (editingItemIndex !== null) { bin.items[editingItemIndex] = { name: name, date: bin.items[editingItemIndex].date || dateStr, photo: photo }; } else { bin.items.push({ name: name, date: dateStr, photo: photo }); } // 2. Persist to storage saveToStorage(); // 3. Immediately close modal and reset form closeModal('addItemModal'); saveBtn.innerText = 'Save Item'; saveBtn.disabled = false; editingItemIndex = null; currentPhotoData = null; // 4. Update UI screens renderItems(bin); renderBins(); // 5. Show toast feedback showToast(editingItemIndex !== null ? '✓ Item Updated!' : '✓ Item Saved!'); }; if (photoFile) { compressImage(photoFile, function(compressedBase64) { finishSave(compressedBase64 || currentPhotoData); }); } else { finishSave(currentPhotoData); } }
